## 1. 背景与动机

原底盘设计使用三种电机驱动配置：

| 预设 | 宏值 | 电机1（转向） | 电机2（动力） | 应用场景 |
|------|------|-------------|-------------|---------|
| `MOTOR_CFG_DEFAULT` | 1 | TB6612 | TB6612 | 原有底盘兼容 |
| `MOTOR_CFG_NEW` | 2 | TB6612 | IBT-4 (BTS7960) | 新动力电机测试 |
| `MOTOR_CFG_2IBT4` | **3** | **IBT-4** | **IBT-4** | **双 IBT-4（本次新增）** |

IBT-4 (BTS7960) 驱动模块相比 TB6612 具有更高电流承载能力（43A vs 1.2A），适合大功率电机驱动场景。新增 `MOTOR_CFG_2IBT4` 预设的动机：

1. 转向电机也需要大电流驱动能力以适应重载底盘
2. 两电机统一使用 IBT-4，简化备件和维护
3. 验证同一颗 TIM1 定时器同时驱动两路 IBT-4 的可行性（4 个独立 PWM 通道全部利用）

## 2. 配置系统架构

### 2.1 文件结构

```
App/config/
├── app_config.h                    ← 顶层选择器：MOTOR_CFG_SET 选择预设
├── app_motor_cfg_default.h         ← 预设 DEFAULT (1)：双 TB6612
├── app_motor_cfg_new.h             ← 预设 NEW     (2)：TB6612 + IBT-4
└── app_motor_cfg_2ibt4.h           ← 预设 2IBT4  (3)：双 IBT-4  ← 新增
```

### 2.2 选择机制

```c
#define MOTOR_CFG_DEFAULT  1
#define MOTOR_CFG_NEW      2
#define MOTOR_CFG_2IBT4    3

#define MOTOR_CFG_SET   3            // ← 唯一编译开关

#if   MOTOR_CFG_SET == MOTOR_CFG_DEFAULT
  #include "app_motor_cfg_default.h"
#elif MOTOR_CFG_SET == MOTOR_CFG_NEW
  #include "app_motor_cfg_new.h"
#elif MOTOR_CFG_SET == MOTOR_CFG_2IBT4
  #include "app_motor_cfg_2ibt4.h"
#else
  #error "MOTOR_CFG_SET must be MOTOR_CFG_DEFAULT (1), MOTOR_CFG_NEW (2), or MOTOR_CFG_2IBT4 (3)"
#endif
```

- `MOTOR_CFG_SET` 是**唯一编译开关**，改一个宏即可切换整套电机配置
- 每个预设文件**自包含**（驱动类型、引脚、PID、编码器等），互不干扰
- 全局参数（`SPEED_LOGIC_MAX`、`PWM_MAX`）定义在 `app_config.h`，所有预设共享

### 2.3 编译时决策链

```
app_config.h:MOTOR_CFG_SET = 3
         │
         ▼
  app_motor_cfg_2ibt4.h        ← 定义 MOTOR1_DRIVER、MOTOR2_DRIVER 等宏
         │
         ▼
  Core/Src/freertos.c          ← #if/#elif 编译期选择 Task 函数
         │
         ├─ Start_MotorControl ──→ IBT4_DC_Task(argument)    ← 电机1
         └─ Start_MotorControl1_T → IBT4_DC_Task(argument)   ← 电机2
         │
         ▼
  App/tasks/ibt4_DC_task.c     ← #if guard 条件编译
         │
         ▼
  App/drivers/motor_DC_IBT4.c  ← 硬件层 HAL PWM 操作
```

### 4.3 任务分派：`freertos.c`

> **路径**: `Core/Src/freertos.c`
> **函数**: `Start_MotorControl` — 电机1 的 FreeRTOS 任务入口
> **变更**: 增加 `#elif MOTOR1_DRIVER == MOTOR_DRIVER_IBT4`

```c
void Start_MotorControl(void *argument)
{
  /* USER CODE BEGIN Start_MotorControl */

  #if MOTOR1_DRIVER == MOTOR_DRIVER_TB6612
  TB6612_DC_Task(argument);
  #elif MOTOR1_DRIVER == MOTOR_DRIVER_IBT4
  IBT4_DC_Task(argument);           /* ← [fix0609] 新增 IBT4 分支 */
  #else
  #error "No valid driver selected for motor1! Check MOTOR1_DRIVER"
  #endif

  /* USER CODE END Start_MotorControl */
}
```

**设计要点**：
- `#if` 预处理指令做**编译时分派**，零运行时开销
- 被调函数（`TB6612_DC_Task` / `IBT4_DC_Task`）本身也是条件编译的，只有需要的符号才被链接
- `#error` 作为安全兜底，防止未定义的驱动类型通过编译
- 新增驱动类型只需增加一个 `#elif` 分支，完全解耦

### 4.4 双电机支持：`ibt4_DC_task.c`

> **路径**: `App/tasks/ibt4_DC_task.c`
> **变更**:
> 1. `#if` 守卫从 `(MOTOR2_DRIVER == MOTOR_DRIVER_IBT4)` 改为 `(MOTOR1_DRIVER == MOTOR_DRIVER_IBT4) || (MOTOR2_DRIVER == MOTOR_DRIVER_IBT4)`
> 2. 硬件初始化通过 `idx == 0` 动态选择 `MOTOR1_IBT4_*` 或 `MOTOR2_IBT4_*` 宏
> 3. 默认指针从 `&g_motors[1]` 修正为 `&g_motors[0]`（详见第5章）

```c
/* 允许两电机分别使用 IBT4，也可混合使用 */
#if (MOTOR1_DRIVER == MOTOR_DRIVER_IBT4) || (MOTOR2_DRIVER == MOTOR_DRIVER_IBT4)

void IBT4_DC_Task(void *argument)
{
    /* ================================================================
     *  [fix0609] 默认指针从 &g_motors[1] → &g_motors[0]
     *  原先只服务于电机2，默认指向 &g_motors[1]；
     *  现在支持双电机路由，当 argument=NULL 时（电机1线程传入）
     *  必须正确默认指向电机1 (&g_motors[0])。
     *  电机2线程显式传入 &g_motors[1]，不受默认值影响。
     * ================================================================ */
    Motor_t *motor = (argument != NULL) ? (Motor_t *)argument : &g_motors[0];
    uint8_t idx = (uint8_t)(motor - &g_motors[0]);

    Motor_PID_Init(motor);      // 按 idx 选 MOTOR1_*/MOTOR2_* PID 参数

    IBT4_Motor_t hw;
    if (idx == 0) {
        IBT4_Motor_Init(&hw, MOTOR1_IBT4_TIM, MOTOR1_IBT4_CH_F, MOTOR1_IBT4_CH_R,
                        MOTOR1_IBT4_EN_PORT, MOTOR1_IBT4_EN_PIN,
                        MOTOR1_MAX_PWM_OUTPUT, MOTOR1_MAX_SPEED_LOGIC,
                        MOTOR1_DEAD_ZONE, MOTOR1_IBT4_POLARITY);
    } else {
        IBT4_Motor_Init(&hw, MOTOR2_IBT4_TIM, MOTOR2_IBT4_CH_F, MOTOR2_IBT4_CH_R,
                        MOTOR2_IBT4_EN_PORT, MOTOR2_IBT4_EN_PIN,
                        MOTOR2_MAX_PWM_OUTPUT, MOTOR2_MAX_SPEED_LOGIC,
                        MOTOR2_DEAD_ZONE, MOTOR2_IBT4_POLARITY);
    }

    /* 按 idx 选队列和互斥锁 */
    osMessageQueueId_t myQueue = (idx == 0) ? MotorQueueHandle : MotorQueue1Handle;
    osMutexId_t        myMutex = (idx == 0) ? motor0_mutexHandle : motor1_mutexHandle;

    CommandMsg_t cmdMsg;

    for (;;)
    {
        /* ---- 命令接收 ---- */
        if (osMessageQueueGet(myQueue, &cmdMsg, NULL, 0) == osOK)
        {
            if (cmdMsg.type == CMD_SET_SPEED || cmdMsg.type == CAN_CMD_SET_SPEED) {
                float new_setpoint = (float)cmdMsg.value;
                if ((new_setpoint > 0.0f && motor->pid.setpoint < 0.0f) ||
                    (new_setpoint < 0.0f && motor->pid.setpoint > 0.0f))
                    PID_Reset(&(motor->pid));
                motor->pid.setpoint = new_setpoint;
            }
            else if (cmdMsg.type == CMD_FORWARD) {
                if (motor->pid.setpoint < 0.0f) PID_Reset(&(motor->pid));
                motor->pid.setpoint = MOTOR_CMD_DEFAULT_SPEED;
            }
            else if (cmdMsg.type == CMD_REVERSE) {
                if (motor->pid.setpoint > 0.0f) PID_Reset(&(motor->pid));
                motor->pid.setpoint = -MOTOR_CMD_DEFAULT_SPEED;
            }
            else if (cmdMsg.type == CMD_STOP || cmdMsg.type == CAN_CMD_STOP) {
                motor->pid.setpoint = 0.0f;
                PID_Reset(&(motor->pid));
            }
        }

        /* ---- PID 闭环控制 ---- */
        if (osMutexAcquire(myMutex, osWaitForever) == osOK)
        {
            motor->target_logic_speed = motor->pid.setpoint;
            if (fabsf(motor->pid.setpoint) > FLT_EPSILON)
            {
                float current_speed = motor->current_logic_speed;

                /* 堵转检测 */
                if (fabsf(current_speed) < 1.0f) {
                    if (motor->stall_counter < 255) motor->stall_counter++;
                    if (motor->stall_counter > 5) motor->flags |= MOTOR_FLAG_STALL;
                } else {
                    motor->stall_counter = 0;
                    motor->flags &= ~MOTOR_FLAG_STALL;
                }

                float output = PID_Compute(&(motor->pid), current_speed);
                IBT4_Motor_SetSpeed(&hw, (int16_t)output);
                motor->pwm_output = hw.pwm_output;

                /* 饱和检测 */
                if (hw.pwm_output >= (PWM_MAX - 10) &&
                    fabsf(current_speed - motor->pid.setpoint) > 10.0f)
                    motor->flags |= MOTOR_FLAG_SATURATED;
                else
                    motor->flags &= ~MOTOR_FLAG_SATURATED;
            }
            else
            {
                IBT4_Motor_Stop(&hw);
                motor->pwm_output = 0;
                motor->stall_counter = 0;
                motor->flags = 0;
            }
            osMutexRelease(myMutex);
        }

        osDelay(10);    // 10ms 控制周期
    }
}

#endif /* MOTOR1_DRIVER == MOTOR_DRIVER_IBT4 || MOTOR2_DRIVER == MOTOR_DRIVER_IBT4 */
```

## 5. Bug 根因分析

### 5.1 症状

切换到 `MOTOR_CFG_2IBT4`（双 IBT4）后：
- 编译通过 ✅
- 动力电机（电机2）正常工作，正反转灵敏 ✅
- 转向电机（电机1）**完全无响应** ❌  — 无声音、无振动、无 PWM 输出
- 示波器测 PA10/PA11 无波形
- 无报错、无异常日志

### 5.2 根因：一行默认指针

**Bug 位置**: `App/tasks/ibt4_DC_task.c` 第 12 行

```c
/* BUG: 默认指向电机2 (idx=1) */
Motor_t *motor = (argument != NULL) ? (Motor_t *)argument : &g_motors[1];
//                                                           ^^^^^^^^^^
//                                                     应当为 &g_motors[0]
```

**触发链**：

```
osThreadNew(Start_MotorControl, NULL, ...)    ← 电机1 线程，传入 NULL
         │
         ▼
Start_MotorControl(NULL)                      ← NULL 参数
         │
         ▼
IBT4_DC_Task(NULL)                            ← argument = NULL
         │
         ▼
motor = &g_motors[1]                          ← BUG: 默认落到电机2！
idx = 1                                       ← 错误地认为是电机2
         │
         ├─ IBT4_Motor_Init(MOTOR2_IBT4_*)   ← 启动了 CH1/CH2 (PA8/PA9)
         ├─ myQueue = MotorQueue1Handle       ← 监听电机2的命令队列
         └─ myMutex = motor1_mutexHandle      ← 使用电机2的互斥锁
```

### 5.3 导致的后果

| 资源 | 期望结果 | 实际结果 |
|------|---------|---------|
| **电机1 PWM 通道** | CH3(PA10)=RPWM, CH4(PA11)=LPWM → 启动 | **从未被初始化，PA10/PA11 无波形** |
| **电机2 PWM 通道** | CH1(PA8)=RPWM, CH2(PA9)=LPWM | 被初始化了 **两次**（电机1 task 误操作 + 电机2 task 正常操作） |
| **电机1 命令队列** | `MotorQueueHandle` 接收转向 CAN ID | 电机1 task 在监听 `MotorQueue1Handle`，**收不到命令** |
| **互斥锁** | `motor0_mutexHandle` | 两 task 都在用 `motor1_mutexHandle`，**相互争抢** |

### 5.4 根因溯源

```c
/* TB6612_DC_Task 的默认设计 */
Motor_t *motor = (argument != NULL) ? (Motor_t *)argument : &g_motors[0];
//                                                           ^^^^^^^^^^
//                                                   设计为电机1 (idx=0)

/* IBT4_DC_Task 的原始默认设计（只支持电机2） */
Motor_t *motor = (argument != NULL) ? (Motor_t *)argument : &g_motors[1];
//                                                           ^^^^^^^^^^
//                                                   设计为电机2 (idx=1)
```

历史原因：`IBT4_DC_Task` 最初**只为电机2服务**，电机2的线程 `Start_MotorControl1_T` 总是显式传入 `&g_motors[1]`，因此默认值 `&g_motors[1]` 从未被触发，只是一个安全 fallback。

当双 IBT4 改造后，电机1 的线程 `Start_MotorControl` 传入 `NULL`，这时默认值被触发——但指向了错误的电机。由于两个 task 函数的默认指针**约定不一致**，这种差异在混合配置时不会暴露，在双 IBT4 时才显现。

### 5.5 为什么编译通过但运行不正常

| 原因 | 说明 |
|------|------|
| **类型系统无保护** | `Motor_t*` 指针类型相同，传给 `idx` 计算在语法上完全合法 |
| **条件编译放行** | `#if (MOTOR1_DRIVER == MOTOR_DRIVER_IBT4)` 为 true，文件正常编译 |
| **无运行时断言** | `idx` 无合法性校验，错误的 idx 不会触发任何错误信号 |
| **部分功能掩盖** | 电机2 正常工作让人误以为配置正确，实际上电机1 在"借尸还魂" |

### 5.6 修复

```diff
-    Motor_t *motor = (argument != NULL) ? (Motor_t *)argument : &g_motors[1];
+    Motor_t *motor = (argument != NULL) ? (Motor_t *)argument : &g_motors[0];
```

### 5.7 修复后验证

| 线程 | 传入参数 | `motor` 指向 | `idx` | 使用的宏 |
|------|---------|-------------|-------|---------|
| `Start_MotorControl` (电机1) | NULL | `&g_motors[0]` ✅ | 0 | `MOTOR1_IBT4_*` |
| `Start_MotorControl1_T` (电机2) | `&g_motors[1]` | `&g_motors[1]` ✅ | 1 | `MOTOR2_IBT4_*` |

### 5.8 经验教训

1. **默认指针的约定应当跨函数一致**：所有 `*_DC_Task` 函数的默认电机指针应当统一为 `&g_motors[0]`，由显式参数处理特例
2. **新增功能时检查所有被调函数的前提假设**：`IBT4_DC_Task` 原本假设自己"只服务电机2"，这个假设在扩展后不成立
3. **NULL 参数路径是隐藏的陷阱**：`osThreadNew(func, NULL, ...)` 传入 NULL 是常见模式，被调函数的默认值必须与之匹配

### 6.5 IBT4 驱动层内部逻辑

```c
// IBT4_Motor_SetSpeed 内部流程：
int16_t speed = output;

// 1. 限幅
if (speed >  MaxSpeed) speed =  MaxSpeed;   // MaxSpeed = 100
if (speed < -MaxSpeed) speed = -MaxSpeed;

// 2. 极性反转 (Polarity=0 时不操作)
if (Polarity) speed = -speed;

// 3. 死区处理 (DeadZone=10)
if (speed > 0 && speed < DeadZone)  speed = DeadZone;
if (speed < 0 && speed > -DeadZone) speed = -DeadZone;

// 4. PWM 值映射
pwmVal = |speed| × MaxPWM / MaxSpeed;   // MaxPWM=7200
// 例: speed=50 → pwmVal = 50×7200/100 = 3600 (50% duty)

// 5. 写入寄存器
if (speed > 0)  → CCR_F = pwmVal, CCR_R = 0    // 正转
if (speed < 0)  → CCR_F = 0,      CCR_R = pwmVal // 反转
if (speed == 0) → CCR_F = 0,      CCR_R = 0      // 停止
```

```c
// Encoder_Task 每 10ms:
```

## 8. IBT-4 驱动详解

### 8.1 BTS7960 芯片简介

BTS7960 是 Infineon 的高电流半桥驱动芯片，IBT-4 模块将两颗 BTS7960 背对背组成全桥：

| 参数 | 值 |
|------|-----|
| 最大持续电流 | 43A |
| 峰值电流 | 50A+ |
| 逻辑电平 | 3.3V / 5V 兼容 |
| Rds(on) | 7mΩ（典型） |
| 开关频率 | 0-25kHz（推荐 ≤10kHz） |
| 保护 | 过流、过温、欠压 |
| 每电机需要 | 2×PWM + 2×EN（可选硬件上拉） |

### 8.2 模块接口定义

| 模块引脚 | IBT-4 丝印 | 方向 | 说明 | 本设计连接 |
|---------|-----------|------|------|-----------|
| **RPWM** | RPWM | 输入 | 正转 PWM（高有效） | 电机1→PA10(TIM1_CH3), 电机2→PA8(TIM1_CH1) |
| **LPWM** | LPWM | 输入 | 反转 PWM（高有效） | 电机1→PA11(TIM1_CH4), 电机2→PA9(TIM1_CH2) |
| **R_EN** | R_EN | 输入 | 正转使能（高有效） | 悬空（硬件上拉） |
| **L_EN** | L_EN | 输入 | 反转使能（高有效） | 悬空（硬件上拉） |
| **VCC** | VCC | 电源 | 逻辑电源（5V） | 来自 STM32 板 |
| **GND** | GND | 电源 | 地 | 共地 |
| **B+** | B+/M | 电源 | 电机电源（6-27V） | 外部稳压电源 |
| **MOTOR** | M | 输出 | 接电机两线 | 正反接口 |

### 8.3 驱动结构体 `IBT4_Motor_t`

```c
typedef struct {
    TIM_HandleTypeDef *htim;           // PWM 定时器句柄
    uint32_t Channel_Forward;          // 正转 PWM 通道 (RPWM)
    uint32_t Channel_Reverse;          // 反转 PWM 通道 (LPWM)
    GPIO_TypeDef *EN_Port;             // 使能端口 (NULL=未连接)
    uint16_t EN_Pin;                   // 使能引脚
    uint16_t MaxPWM;                   // 最大 PWM 值 (TIM ARR)
    int16_t  MaxSpeed;                 // 最大逻辑速度
    uint16_t DeadZone;                 // 死区 (低于此值 PWM=0)
    uint8_t  Polarity;                 // 极性 (0=正常, 1=反转)
    int16_t  pwm_output;               // 当前 PWM 输出值 (调试用)
} IBT4_Motor_t;
```

### 8.4 驱动 API

```c
/* 初始化：配置通道、启动 PWM、设置初始占空比 0 */
void IBT4_Motor_Init(IBT4_Motor_t *motor,
                     TIM_HandleTypeDef *htim,
                     uint32_t Channel_Forward,
                     uint32_t Channel_Reverse,
                     GPIO_TypeDef *EN_Port, uint16_t EN_Pin,
                     uint16_t MaxPWM, int16_t MaxSpeed,
                     uint16_t DeadZone, uint8_t Polarity);

/* 设置速度：限幅→极性→死区→PWM映射→写入CCR */
void IBT4_Motor_SetSpeed(IBT4_Motor_t *motor, int16_t speed);

/* 停止：CH_F=0, CH_R=0 */
void IBT4_Motor_Stop(IBT4_Motor_t *motor);
```

### 8.5 控制逻辑

```
speed > 0  → CH_F(RPWM) = pwm,   CH_R(LPWM) = 0   ← 正转
speed < 0  → CH_F(RPWM) = 0,     CH_R(LPWM) = pwm ← 反转
speed = 0  → CH_F(RPWM) = 0,     CH_R(LPWM) = 0   ← 停止
```

各预设实际通道映射：

| 预设 | 电机 | RPWM(CH_F) | LPWM(CH_R) |
|------|------|------------|------------|
| NEW (2) | 电机2 | TIM1_CH2 (PA9) | TIM1_CH3 (PA10) |
| 2IBT4 (3) | 电机1 | TIM1_CH3 (PA10) | TIM1_CH4 (PA11) |
| 2IBT4 (3) | 电机2 | TIM1_CH1 (PA8) | TIM1_CH2 (PA9) |

### 8.6 PWM 值计算

```c
pwmVal = (|speed| × MaxPWM) / MaxSpeed;

// MaxPWM = 7200 (TIM1 ARR), MaxSpeed = 100 (SPEED_LOGIC_MAX)
// speed = 50  → pwmVal = 50 × 7200 / 100 = 3600  (50% duty)
// speed = 100 → pwmVal = 100 × 7200 / 100 = 7200  (100% duty)
// speed = 0   → pwmVal = 0                        (停止)
```

### 8.7 TB6612 vs IBT4 对比

| 特性 | TB6612 | IBT-4 (BTS7960) |
|------|--------|-----------------|
| **最大持续电流** | 1.2A | 43A |
| **峰值电流** | 3.2A (20ms) | 50A+ |
| **Rds(on)** | 0.5Ω (典型) | 7mΩ (典型) |
| **逻辑电平** | 2.7-5.5V | 3.3-5V |
| **控制信号** | 1×PWM + 2×方向GPIO | 2×PWM (RPWM/LPWM) |
| **停止模式** | 刹车/滑行可选 | 刹车（PWM=0） |
| **使能引脚** | 可选 (EN) | 双 EN (建议硬件上拉) |
| **适用场景** | 小功率电机 (<10W) | 大功率电机 (>10W) |
| **所需宏数量** | ~8个 MOTORx_* 宏 | ~12个 MOTORx_* 宏 |
| **PWM 通道占用** | 每电机 1 通道 | 每电机 2 通道 |
| **PCB 面积** | 小 (QFN 封装) | 大 (独立模块) |

### 8.8 IBT4 常见问题排查

| 症状 | 可能原因 | 排查方法 |
|------|---------|---------|
| 电机完全不动 | R_EN/L_EN 未拉高 | 测模块 EN 引脚电压，应为高电平 |
| 电机只往一个方向转 | LPWM/RPWM 有一路没波形 | 示波器测两路 PWM 引脚 |
| 电机抖动不转 | 死区设置过大或 PWM 频率不匹配 | 检查 DeadZone ≤ 20, 降低 TIM 频率 |
| 电机有嗡鸣声但不动 | 供电不足或堵转 | 测电机电源电压, 手动盘轴 |
| 电机发热严重 | PWM 频率太低或死区重叠 | 确保 RPWM/LPWM 不同时为高 |

### 10.5 `App/drivers/motor_DC_IBT4.c` — 驱动完整代码

```c
#include "motor_DC_IBT4.h"

#define IBT4_DEAD_ZONE_MIN  1

void IBT4_Motor_Init(IBT4_Motor_t *motor,
                     TIM_HandleTypeDef *htim,
                     uint32_t Channel_Forward,
                     uint32_t Channel_Reverse,
                     GPIO_TypeDef *EN_Port, uint16_t EN_Pin,
                     uint16_t MaxPWM, int16_t MaxSpeed,
                     uint16_t DeadZone, uint8_t Polarity)
{
    motor->htim            = htim;
    motor->Channel_Forward = Channel_Forward;
    motor->Channel_Reverse = Channel_Reverse;
    motor->EN_Port         = EN_Port;
    motor->EN_Pin          = EN_Pin;
    motor->MaxPWM          = MaxPWM;
    motor->MaxSpeed        = MaxSpeed;
    motor->DeadZone        = (DeadZone < IBT4_DEAD_ZONE_MIN) ? IBT4_DEAD_ZONE_MIN : DeadZone;
    motor->Polarity        = (Polarity > 0) ? 1 : 0;
    motor->pwm_output      = 0;

    if (htim != NULL) {
        HAL_TIM_PWM_Start(htim, Channel_Forward);
        HAL_TIM_PWM_Start(htim, Channel_Reverse);
        __HAL_TIM_SET_COMPARE(htim, Channel_Forward, 0);
        __HAL_TIM_SET_COMPARE(htim, Channel_Reverse, 0);
    }
    if (EN_Port != NULL && EN_Pin != 0) {
        HAL_GPIO_WritePin(EN_Port, EN_Pin, GPIO_PIN_SET);
    }
}

void IBT4_Motor_SetSpeed(IBT4_Motor_t *motor, int16_t speed)
{
    if (speed > motor->MaxSpeed)  speed = motor->MaxSpeed;
    if (speed < -motor->MaxSpeed) speed = -motor->MaxSpeed;
    if (motor->Polarity) speed = -speed;

    if (speed > 0 && speed < (int16_t)motor->DeadZone)  speed = motor->DeadZone;
    if (speed < 0 && speed > -(int16_t)motor->DeadZone) speed = -(int16_t)motor->DeadZone;

    uint16_t pwmVal = 0;
    if (motor->MaxSpeed != 0)
        pwmVal = (uint16_t)(((int32_t)(speed > 0 ? speed : -speed) * motor->MaxPWM) / motor->MaxSpeed);

    if (speed > 0) {
        __HAL_TIM_SET_COMPARE(motor->htim, motor->Channel_Forward, pwmVal);
        __HAL_TIM_SET_COMPARE(motor->htim, motor->Channel_Reverse, 0);
    } else if (speed < 0) {
        __HAL_TIM_SET_COMPARE(motor->htim, motor->Channel_Forward, 0);
        __HAL_TIM_SET_COMPARE(motor->htim, motor->Channel_Reverse, pwmVal);
    } else {
        __HAL_TIM_SET_COMPARE(motor->htim, motor->Channel_Forward, 0);
        __HAL_TIM_SET_COMPARE(motor->htim, motor->Channel_Reverse, 0);
    }
    motor->pwm_output = pwmVal;
}

void IBT4_Motor_Stop(IBT4_Motor_t *motor)
{
    __HAL_TIM_SET_COMPARE(motor->htim, motor->Channel_Forward, 0);
    __HAL_TIM_SET_COMPARE(motor->htim, motor->Channel_Reverse, 0);
    motor->pwm_output = 0;
}
```
