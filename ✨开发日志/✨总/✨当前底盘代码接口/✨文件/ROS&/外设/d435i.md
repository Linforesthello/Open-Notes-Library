> [!NOTE]
> ## 正确启动方式
> 
> 你现在不要直接：
> 
> ```
> ros2 launch realsense2_camera rs_launch.py
> ```
> 
> 而是：
> 
> ## 开启 pointcloud
> 
> 执行：
> 
> ```
> ros2 launch realsense2_camera rs_launch.py pointcloud.enable:=true
> ```
> 
> 这是官方 wrapper 参数。([github.com](https://github.com/IntelRealSense/realsense-ros?utm_source=chatgpt.com))

### 驱动指令
%% 相机驱动 %%
ros2 launch realsense2_camera rs_launch.py     depth_module.depth_profile:=424x240x15     rgb_camera.color_profile:=424x240x15     align_depth.enable:=true     enable_sync:=true

%% rtabmap %%
ros2 run rtabmap_odom rgbd_odometry     --ros-args     -p frame_id:=camera_link     -p approx_sync:=true     -p approx_sync_max_interval:=0.1     -p queue_size:=100     -r rgb/image:=/camera/camera/color/image_raw     -r depth/image:=/camera/camera/depth/image_rect_raw     -r rgb/camera_info:=/camera/camera/color/camera_info     -r odom:=/rtabmap/odom

%% rtabmap %%
ros2 run rtabmap_slam rtabmap     --ros-args     -r rgb/image:=/camera/camera/color/image_raw     -r depth/image:=/camera/camera/depth/image_rect_raw     -r rgb/camera_info:=/camera/camera/color/camera_info     -r odom:=/rtabmap/odom     -p frame_id:=camera_link     -p approx_sync:=true     -p sync_queue_size:=100     -p Grid/FromDepth:=true     -p database_path:=~/.ros/rtabmap_final.db
### 查验指令

ros2 topic hz /camera/camera/color/image_raw

ros2 topic hz /rtabmap/odom

ros2 topic hz /camera/camera/depth/image_rect_raw

ros2 topic hz /camera/camera/color/camera_info
